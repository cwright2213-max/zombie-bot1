def apply_elemental_effect():
    effect = ELEMENTAL_AMMO[current_ammo_type]["effect"]
    if not effect: return None
    rolls = {"bleed": 0.25, "burn": 0.30, "freeze": 0.20, "poison": 0.35, "shock": 0.15}
    if random.random() < rolls.get(effect, 0): return effect
    return None
